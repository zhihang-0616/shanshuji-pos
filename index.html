function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); var action = data.action || "append"; var itemsArr = data.items; if (typeof itemsArr === "string") { try { itemsArr = JSON.parse(itemsArr); } catch(err) { itemsArr = []; } } var itemsStr = itemsArr.map(function(item) { var subText = ""; if (item.subItems && item.subItems.length > 0) { subText = "(" + item.subItems.map(function(s) { return s.name; }).join(", ") + ")"; } return item.name + (item.qty > 1 ? "x" + item.qty : "") + subText; }).join("、"); var flavorStr = ""; var prefs = data.prefs; if (typeof prefs === "string") { try { prefs = JSON.parse(prefs); } catch(err) { prefs = {}; } } if (prefs) { var parts = []; if (prefs.flavor && prefs.flavor !== "") parts.push(prefs.flavor); if (prefs.spice && prefs.spice !== "") parts.push(prefs.spice); if (prefs.salt && prefs.salt !== "") parts.push(prefs.salt); flavorStr = parts.join(" "); } var rowData = [ String(data.id), data.date, data.weekDay, data.time, data.customer, data.phone, data.total, data.payment, data.status || "", itemsStr, flavorStr, data.custNotes ]; var lastRow = sheet.getLastRow(); var foundRowIndex = -1; if (lastRow > 1) { var idValues = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (var i = 0; i < idValues.length; i++) { var sheetId = String(idValues[i][0]).trim(); var targetId = String(data.id).trim(); if (sheetId !== "" && sheetId === targetId) { foundRowIndex = i + 2; break; } } } if (foundRowIndex !== -1) { sheet.getRange(foundRowIndex, 1, 1, rowData.length).setValues([rowData]); } else { sheet.appendRow(rowData); } return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var action = e.parameter.action; var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (action === "exportDailyEXL") { var targetDate = e.parameter.date; // 格式: YYYY/MM/DD var rows = sheet.getDataRange().getValues(); var matchOrders = []; for (var i = 1; i < rows.length; i++) { var r = rows[i]; if (r[1] === targetDate && r[4] !== "【當日總結報表】") { matchOrders.push(r); } } // 橫向展開格式輸出 var html = ""; for (var k = 0; k < 15; k++) { html += ""; } html += ""; matchOrders.forEach(function(ord) { html += ""; var itemsText = ord[9]; // 簡易拆解品項 var items = itemsText.split("、"); items.forEach(function(it) { html += ""; }); html += ""; }); html += "
日期時間總額單品單價口味備註
" + ord[1] + "" + ord[3] + "" + ord[6] + "" + it + "0" + ord[10] + "" + ord[11] + "
"; return HtmlService.createHtmlOutput(html); } var rows = sheet.getDataRange().getValues(); var orders = []; if (rows.length > 1) { for (var i = 1; i < rows.length; i++) { var r = rows[i]; if (!r[0]) continue; orders.push({ id: r[0], date: r[1], weekDay: r[2], time: r[3], customer: r[4], phone: r[5], total: r[6], payment: r[7], status: r[8], items: [{ name: r[9], qty: 1, price: r[6] }], custNotes: r[11] }); } } return ContentService.createTextOutput(JSON.stringify(orders)).setMimeType(ContentService.MimeType.JSON); }
